    # --- MOTOR DE LIMPEZA RÁPIDA (O que estamos mexendo agora) ---
    def get_size(self, start_path):
        total = 0
        try:
            for dirpath, dirnames, filenames in os.walk(start_path):
                for f in filenames:
                    fp = os.path.join(dirpath, f)
                    if os.path.exists(fp): total += os.path.getsize(fp)
        except: pass
        return total

    def format_bytes(self, size):
        for unit in ['B', 'KB', 'MB', 'GB']:
            if size < 1024: return f"{size:.2f} {unit}"
            size /= 1024
        return f"{size:.2f} TB"

    def run_clean_auto(self):
        pastas = [os.environ.get('TEMP'), 'C:/Windows/Temp', 'C:/Windows/Prefetch']
        t_inicial = sum(self.get_size(p) for p in pastas if p and os.path.exists(p))
        
        if t_inicial < 1024 * 1024:
            self.status_lbl.setText("> Sistema já limpo recentemente.")
            self.status_lbl.setStyleSheet("color: #ffbd2e; font-family: 'Consolas';")
            return

        self.status_lbl.setText("> Iniciando limpeza de arquivos...")
        QApplication.processEvents()
        
        cmd = 'del /s /f /q %temp%\\*.* & del /s /f /q C:\\Windows\\Temp\\*.* & del /s /f /q C:\\Windows\\Prefetch\\*.* & ipconfig /flushdns'
        subprocess.run(f'start cmd /c "echo TECNOSUP: LIMPANDO DISCO... & {cmd} & exit"', shell=True)

        QTimer.singleShot(2000, lambda: self.finish_clean(t_inicial))

    def finish_clean(self, t_inicial):
        pastas = [os.environ.get('TEMP'), 'C:/Windows/Temp', 'C:/Windows/Prefetch']
        t_final = sum(self.get_size(p) for p in pastas if p and os.path.exists(p))
        limpo = t_inicial - t_final
        agora = datetime.datetime.now().strftime("%d/%m/%Y às %H:%M:%S")
        with open(self.log_file, "w") as f: f.write(agora)
        
        self.status_lbl.setText(f"> SUCESSO: {self.format_bytes(limpo)} removidos!\n> Última vez: {agora}")
        self.status_lbl.setStyleSheet(f"color: {self.primary}; font-family: 'Consolas'; font-weight: bold;")

    def add_action_btn(self, text, func):

    def add_title(self, text, color=None):

        self.content_lyt.addWidget(title, alignment=Qt.AlignCenter); self.content_lyt.addSpacing(20)

    def add_status_bar(self, msg="> Aguardando comando..."):

    def paint_grid(self, event):
        painter = QPainter(self.content_container)
        pen = QPen(QColor(14, 179, 255, 12)); painter.setPen(pen)
        for x in range(0, self.content_container.width(), 40): painter.drawLine(x, 0, x, self.content_container.height())
        for y in range(0, self.content_container.height(), 40): painter.drawLine(0, y, self.content_container.width(), y)

    def add_neon(self, widget, color):
        glow = QGraphicsDropShadowEffect()
        glow.setBlurRadius(20); glow.setColor(QColor(color)); glow.setOffset(0)
        widget.setGraphicsEffect(glow)

    def create_menu_btn(self, text, func):
        btn = QPushButton(text); btn.setObjectName("MenuBtn"); btn.setFixedSize(190, 40)
        btn.setCursor(QCursor(Qt.PointingHandCursor)); btn.clicked.connect(func)
        return btn

    def clear_screen(self):
        while self.content_lyt.count():
            item = self.content_lyt.takeAt(0); 
            if item.widget(): item.widget().deleteLater()

    def show_home(self):
